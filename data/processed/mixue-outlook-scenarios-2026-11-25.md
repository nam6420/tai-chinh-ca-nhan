# Đơn vị tiền tệ là triệu RMB, trừ khi chú thích.
cash_equivalent = 6027.942
deposit_short, deposit_long, cash_restricted = 5798.695, 1716.561, 11.488
cash_total_management, fvtpl = 21640.7, 8096.059
wealth_products = cash_total_management - (
    cash_equivalent + deposit_short + deposit_long + cash_restricted
)
shares = 379.6188  # triệu cổ phiếu
market_cap = 155.70 * shares * 0.85587
special_dividend = 2.65 * shares
non_controlling_interest = 332.336
lease_liabilities = 94.076 + 133.674
parent_profit_ltm = 5886.704 - 2692.884 + 2296.314

operating_cash_flow = 6039.033 - 3275.383 + 2049.818
capex = 685.130 - 305.841 + 289.367 + 6.358 + 38.409
lease_principal = 91.705 - 35.689 + 54.285
lease_interest = 4.874 - 2.072 + 3.163
interest_received = 135.412 - 42.914 + 58.262
free_cash_flow = (
    operating_cash_flow - capex - lease_principal - lease_interest
    - interest_received * (1 - 0.215)
)

def excess_cash(operating_reserve, wealth_haircut, include_long_deposits):
    return (
        cash_equivalent + deposit_short + deposit_long * include_long_deposits
        + wealth_products * (1 - wealth_haircut)
        - special_dividend - operating_reserve
    )

def implied_growth(enterprise_value, cash_flow, cost_of_capital):
    return (enterprise_value * cost_of_capital - cash_flow) / (enterprise_value + cash_flow)

cash_cases = {
    "Rộng": (0, 0, True),
    "Minh họa": (2000, 0.25, True),
    "Chặt": (2000, 1, False),
}
print("Vốn hóa, cổ tức, P/E, FCF, sản phẩm quản lý tài sản")
print(market_cap, special_dividend, market_cap / parent_profit_ltm, free_cash_flow, wealth_products)
for label, settings in cash_cases.items():
    surplus = excess_cash(*settings)
    enterprise_value = market_cap + non_controlling_interest - surplus
    print(label, round(surplus, 1), round(enterprise_value, 1),
          [round(implied_growth(enterprise_value, free_cash_flow, wacc) * 100, 2)
           for wacc in (0.08, 0.10, 0.12)])
    print("FCF tại chi phí vốn 10%:", [
        round(implied_growth(enterprise_value, cash_flow, 0.10) * 100, 2)
        for cash_flow in (3500, free_cash_flow, 4500)
    ])

stores_2024, stores_2025, revenue_2025, profit_2025 = 46479, 59823, 33559.923, 5927.077
stores_h1_2025, stores_h1_2026 = 53014, 63987
sales_h1_2025, sales_h1_2026 = 14874.809, 15215.804
sales_h2_2025 = revenue_2025 - sales_h1_2025
stores_h2_2025_avg = (stores_h1_2025 + stores_2025) / 2
scenarios = {
    "Suy giảm sâu": ((6000, -3000, -6000), (-.20, -.15, -.10), .09),
    "Bất lợi": ((7000, 2000, -1000), (-.17, -.08, -.03), .13),
    "Cơ sở": ((8000, 5000, 3500), (-.15, -.02, 0), .15),
    "Thuận lợi": ((9000, 6500, 5500), (-.12, .05, .03), .165),
}
for label, (store_changes, rps_changes, margin_2028) in scenarios.items():
    stores = stores_2025
    average_store_revenue = revenue_2025 / ((stores_2024 + stores_2025) / 2)
    annual_revenues = []
    for change, rps_change in zip(store_changes, rps_changes):
        closing_stores = stores + change
        average_store_revenue *= 1 + rps_change
        annual_revenues.append((stores + closing_stores) / 2 * average_store_revenue)
        stores = closing_stores
    stores_h2_2026_avg = (stores_h1_2026 + stores_2025 + store_changes[0]) / 2
    sales_h2_2026 = annual_revenues[0] - sales_h1_2026
    h2_sales_growth = sales_h2_2026 / sales_h2_2025 - 1
    h2_rps_growth = (
        (sales_h2_2026 / stores_h2_2026_avg)
        / (sales_h2_2025 / stores_h2_2025_avg) - 1
    )
    revenue_2028 = annual_revenues[-1]
    profit_2028 = revenue_2028 * margin_2028
    print(label, stores, [round(value, 1) for value in annual_revenues],
          round(h2_sales_growth * 100, 1), round(h2_rps_growth * 100, 1),
          round(profit_2028, 1), round(((revenue_2028 / revenue_2025) ** (1/3) - 1) * 100, 1),
          round(((profit_2028 / profit_2025) ** (1/3) - 1) * 100, 1))

base_revenue, base_margin = 39222.6, .15
print("Sự cố minh họa", base_revenue * .90 * (base_margin - .02))
